`); }); // Criar checkout app.post("/pagamento", async(req,res)=>{ const preference = new Preference(client); const pagamento = await preference.create({ body:{ items:[ { title:"Produto Exemplo", quantity:1, unit_price:100 } ], back_urls:{ success:"https://seusite.com/sucesso", failure:"https://seusite.com/erro" } } }); res.json({ link: pagamento.init_point }); }); app.listen(3000,()=>{ console.log("Loja rodando em http://localhost:3000"); }); npm init -y npm install express cors mercadopago node app.js http://localhost:3000
